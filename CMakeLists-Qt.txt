cmake_minimum_required(VERSION 3.16)
project(ClipBridge VERSION 2.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

# 查找 Qt 包
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)

# 源文件分组
set(SOURCES
    src-qt/main.cpp
    src-qt/core/actionmanager.cpp
    src-qt/core/appconfig.cpp
    src-qt/ui/trayicon.cpp
    src-qt/utils/clipboardhelper.cpp
    src-qt/utils/hotkey.cpp
)

set(HEADERS
    src-qt/core/actionmanager.h
    src-qt/core/appconfig.h
    src-qt/ui/trayicon.h
    src-qt/utils/clipboardhelper.h
    src-qt/utils/hotkey.h
)

# 创建可执行文件
add_executable(ClipBridge
    ${SOURCES}
    ${HEADERS}
)

# 链接 Qt 库
target_link_libraries(ClipBridge PRIVATE
    Qt6::Core
    Qt6::Gui
    Qt6::Widgets
)

# 包含目录
target_include_directories(ClipBridge PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src-qt
    ${CMAKE_CURRENT_SOURCE_DIR}/src-qt/core
    ${CMAKE_CURRENT_SOURCE_DIR}/src-qt/ui
    ${CMAKE_CURRENT_SOURCE_DIR}/src-qt/utils
)

# Windows 特定配置
if(WIN32)
    set_target_properties(ClipBridge PROPERTIES
        WIN32_EXECUTABLE ON
    )
endif()

# 复制配置文件
add_custom_command(TARGET ClipBridge POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/config.json"
    $<TARGET_FILE_DIR:ClipBridge>/config.json
)
